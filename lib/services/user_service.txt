import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:syncskills/models/user.dart';

/// Minimal UserService that uses Firestore. It provides basic
/// get/save operations used by the controllers and registered in main.dart.
class UserService {
  final CollectionReference<Map<String, dynamic>> _users = FirebaseFirestore
      .instance
      .collection('users');

  Future<User?> getUser(String id) async {
    final doc = await _users.doc(id).get();
    if (!doc.exists) return null;
    return User.fromFirestore(doc);
  }

  Future<void> saveUser(User user) async {
    await _users.doc(user.id).set(user.toFirestore(), SetOptions(merge: true));
  }

  Future<String> createUser(User user) async {
    final doc = await _users.add(user.toFirestore());
    return doc.id;
  }

  Future<void> deleteUser(String id) async {
    await _users.doc(id).delete();
  }
}
